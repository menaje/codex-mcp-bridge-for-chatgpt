async function request(label, path, body) {
  const beganAt = new Date().toISOString();
  const bytes = body === undefined ? undefined : JSON.stringify(body);
  try {
    const response = await fetch(new URL(path, input.origin), {
      method: bytes === undefined ? 'GET' : 'POST', redirect: 'manual',
      headers: {
        'OAI-Sites-Authorization': `Bearer ${input.token}`,
        'Accept': 'application/json, text/event-stream',
        'Cache-Control': 'no-store', 'Connection': 'close',
        ...(bytes === undefined ? {} : { 'Content-Type': 'application/json' }),
      }, body: bytes, signal: AbortSignal.timeout(20000),
    });
    const raw = (await response.text()).split(input.token).join('[REDACTED]');
    const record = {
      label, path, method: bytes === undefined ? 'GET' : 'POST', beganAt,
      completedAt: new Date().toISOString(), status: response.status,
      responseDate: response.headers.get('date'), contentType: response.headers.get('content-type'),
      payloadSha256: bytes === undefined ? null : digest(bytes),
      responseSha256: digest(raw), responseBody: raw,
    };
    report.requests.push(record);
    console.log(JSON.stringify({ label, status: response.status, completedAt: record.completedAt }));
    return record;
  } catch {
    const record = { label, path, beganAt, completedAt: new Date().toISOString(), status: null, error: 'HTTP request failed or timed out' };
    report.requests.push(record);
    console.log(JSON.stringify(record));
    return record;
  }
}
