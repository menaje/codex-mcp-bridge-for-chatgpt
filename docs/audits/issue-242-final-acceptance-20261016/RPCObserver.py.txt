 async def handle(self,reader,writer):
  task=asyncio.current_task();self.tasks.add(task);upwriter=None;method='unparsed';started=time.monotonic()
  try:
   upstream,upwriter=await asyncio.open_unix_connection(str(self.backend),limit=16*1024*1024)
   # Native protocol sends one newline-framed request, then one response.
   line=await reader.readline();req=json.loads(line);method=req.get('method','invalid')
   if not isinstance(method,str) or len(method)>100:method='invalid'
   self.emit({'kind':'request','lane':self.lane,'method':method})
   upwriter.write(line);await upwriter.drain()
   line=await upstream.readline()
   # Forward exactly once and never retry. Save only approved aggregate fields.
   writer.write(line);await writer.drain();reply=json.loads(line);result=reply.get('result')
   event={'kind':'response','lane':self.lane,'method':method,'elapsedMs':round((time.monotonic()-started)*1000,3),'ok':'error' not in reply}
   if method=='changes.wait' and isinstance(result,dict):
    event['topics']=[t for t in result.get('topics',[]) if t in TOPICS]
    event['completionTopicSupported']='completion-outbox-ready' in result.get('supportedTopics',[])
   if method=='completion.availability' and isinstance(result,dict):event['available']=result.get('available')
   if method=='completion.claim' and isinstance(result,dict):event['eventCount']=len(result.get('events',[]))
   self.emit(event)
  except (Exception,asyncio.CancelledError) as e:self.emit({'kind':'transport-error','lane':self.lane,'method':method,'type':type(e).__name__})
  finally:
   for w in [upwriter,writer]:
    if w:
     w.close()
     try:await w.wait_closed()
     except Exception:pass
   self.tasks.discard(task)
